cmake_minimum_required(VERSION 3.14)
project(opentpu_ramulator_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT RAMULATOR_ROOT)
  message(FATAL_ERROR "Set RAMULATOR_ROOT to the pinned, prepared and built Ramulator clone")
endif()
add_library(opentpu_ramulator SHARED bridge.cpp)
target_include_directories(opentpu_ramulator PRIVATE ${RAMULATOR_ROOT}/src ${RAMULATOR_ROOT}/ext/yaml-cpp/include ${RAMULATOR_ROOT}/ext/spdlog/include)
find_library(RAMULATOR_LIBRARY ramulator PATHS ${RAMULATOR_ROOT} NO_DEFAULT_PATH REQUIRED)
target_link_libraries(opentpu_ramulator PRIVATE ${RAMULATOR_LIBRARY})
set_target_properties(opentpu_ramulator PROPERTIES BUILD_RPATH ${RAMULATOR_ROOT})
